cmake_minimum_required(VERSION 3.16)
project(era_ito LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ITO_SOURCE)
  message(FATAL_ERROR "Set ITO_SOURCE to the pinned upstream checkout")
endif()
add_library(ito-core STATIC ${ITO_SOURCE}/esp32/engine/itofs.c ../tempo/sonic.c)
target_include_directories(ito-core PUBLIC ${ITO_SOURCE}/esp32/engine ../tempo)
target_compile_options(ito-core PRIVATE -O2 -ffp-contract=off)
add_executable(ito-speak cli.cpp)
target_compile_options(ito-speak PRIVATE -O2)
target_link_options(ito-speak PRIVATE -static -s)
target_link_libraries(ito-speak PRIVATE ito-core m)
